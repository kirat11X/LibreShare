/// Windows DPAPI implementation of [SecretStore] using `CryptProtectData` and
/// `CryptUnprotectData` via `package:win32`.
///
/// NOTE: This file is delivered as reference code (.dart.txt) in
/// `docs/plans/f-proposals/F5/` to avoid adding `package:win32` to the Linux/macOS
/// build tree or failing analyzer before the orchestrator approves the dependency.
///
/// Security properties:
/// - Secrets are encrypted using the Windows Data Protection API (DPAPI).
/// - Encryption is tied to the current Windows user login credentials.
/// - Atomic file writes (via temp file and rename) prevent half-written blobs.
/// - Corruption or decryption failure throws [SecretStoreException], never returning
///   null or regenerating identities.
/// - Zero secret logging: secret names and values are never included in log records
///   or exception strings.
library;

import 'dart:convert';
import 'dart:ffi';
import 'dart:io';
import 'dart:typed_data';

import 'package:crypto/crypto.dart';
import 'package:ffi/ffi.dart';
import 'package:logging/logging.dart';
import 'package:win32/win32.dart';
import 'package:libreshare_core/libreshare_core.dart';

final Logger _log = Logger('libreshare.platform.secret.dpapi');

/// A Windows DPAPI-backed [SecretStore].
class DpapiSecretStore implements SecretStore {
  DpapiSecretStore(this.directory);

  /// Directory where encrypted DPAPI secret files are stored.
  final Directory directory;

  static const String fileExtension = '.dpapi';
  static const int maxValueBytes = 512 * 1024; // 512 KiB matching FileSecretStore

  String _fileNameFor(String name) {
    if (!isValidSecretName(name)) {
      throw ArgumentError('invalid secret name');
    }
    final hash = sha256.convert(utf8.encode(name)).toString();
    return '$hash$fileExtension';
  }

  File _fileFor(String name) {
    return File('${directory.path}${Platform.pathSeparator}${_fileNameFor(name)}');
  }

  @override
  Future<Uint8List?> get(String key) async {
    final file = _fileFor(key);
    if (!file.existsSync()) {
      return null;
    }

    Uint8List encrypted;
    try {
      encrypted = await file.readAsBytes();
    } on Object catch (e) {
      _log.warning('Could not read DPAPI secret file: ${e.runtimeType}');
      throw const SecretStoreException('secret store file is unreadable');
    }

    if (encrypted.isEmpty) {
      throw const SecretStoreException('corrupted protected blob: empty file');
    }

    return _unprotect(encrypted);
  }

  @override
  Future<void> put(String key, Uint8List value) async {
    if (value.length > maxValueBytes) {
      throw ArgumentError('secret value too large');
    }

    if (!directory.existsSync()) {
      await directory.create(recursive: true);
    }

    final encrypted = _protect(value);
    final target = _fileFor(key);
    final tmp = File('${target.path}.tmp');

    try {
      await tmp.create();
      final raf = await tmp.open(mode: FileMode.write);
      try {
        await raf.writeFrom(encrypted);
        await raf.flush();
      } finally {
        await raf.close();
      }
      await tmp.rename(target.path);
    } on Object catch (e) {
      try {
        if (tmp.existsSync()) await tmp.delete();
      } on Object {}
      _log.warning('Failed to write DPAPI secret: ${e.runtimeType}');
      throw const SecretStoreException('cannot write protected secret');
    }
  }

  @override
  Future<void> delete(String key) async {
    final file = _fileFor(key);
    if (file.existsSync()) {
      try {
        await file.delete();
      } on Object catch (e) {
        _log.warning('Failed to delete DPAPI secret: ${e.runtimeType}');
      }
    }
  }

  Uint8List _protect(Uint8List plaintext) {
    return using((Arena arena) {
      final pDataIn = arena<DATA_BLOB>();
      final pDataOut = arena<DATA_BLOB>();

      final inBytes = arena<Uint8>(plaintext.length);
      inBytes.asTypedList(plaintext.length).setAll(0, plaintext);

      pDataIn.ref.cbData = plaintext.length;
      pDataIn.ref.pbData = inBytes;

      final result = CryptProtectData(
        pDataIn,
        nullptr, // szDataDescr
        nullptr, // pOptionalEntropy
        nullptr, // pvReserved
        nullptr, // pPromptStruct
        CRYPTPROTECT_UI_FORBIDDEN,
        pDataOut,
      );

      if (result == 0) {
        final err = GetLastError();
        _log.severe('CryptProtectData failed with error $err');
        throw SecretStoreException('CryptProtectData failed with error $err');
      }

      try {
        final outputBytes = Uint8List.fromList(
          pDataOut.ref.pbData.asTypedList(pDataOut.ref.cbData),
        );
        return outputBytes;
      } finally {
        LocalFree(pDataOut.ref.pbData.address);
      }
    });
  }

  Uint8List _unprotect(Uint8List ciphertext) {
    return using((Arena arena) {
      final pDataIn = arena<DATA_BLOB>();
      final pDataOut = arena<DATA_BLOB>();

      final inBytes = arena<Uint8>(ciphertext.length);
      inBytes.asTypedList(ciphertext.length).setAll(0, ciphertext);

      pDataIn.ref.cbData = ciphertext.length;
      pDataIn.ref.pbData = inBytes;

      final result = CryptUnprotectData(
        pDataIn,
        nullptr, // ppszDataDescr
        nullptr, // pOptionalEntropy
        nullptr, // pvReserved
        nullptr, // pPromptStruct
        CRYPTPROTECT_UI_FORBIDDEN,
        pDataOut,
      );

      if (result == 0) {
        final err = GetLastError();
        _log.severe('CryptUnprotectData failed with error $err');
        throw SecretStoreException('CryptUnprotectData failed with error $err');
      }

      try {
        final outputBytes = Uint8List.fromList(
          pDataOut.ref.pbData.asTypedList(pDataOut.ref.cbData),
        );
        return outputBytes;
      } finally {
        LocalFree(pDataOut.ref.pbData.address);
      }
    });
  }
}
