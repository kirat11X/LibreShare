// VERIFIER experiment (round 2): does destroy() of a Socket that SecureSocket.secureServer has
// taken over reach the TCP connection? Not a deliverable test.
import 'dart:async';
import 'dart:io';

import 'package:libreshare_core/libreshare_core.dart';
import 'package:test/test.dart';

import '../helpers/p3_secure_rig.dart';

Future<bool> clientSeesClose(Socket c, Duration t) async {
  final done = Completer<void>();
  c.listen((_) {}, onDone: done.complete, onError: (Object _) => done.complete());
  try {
    await done.future.timeout(t);
    return true;
  } on TimeoutException {
    return false;
  }
}

void main() {
  for (final how in ['destroy', 'close']) {
    test('E1 Socket + secureServer, then $how() on the original socket after 1 s', () async {
      final raw = await ServerSocket.bind(InternetAddress.loopbackIPv4, 0);
      addTearDown(raw.close);
      var hsOutcome = 'pending';
      raw.listen((s) {
        SecureSocket.secureServer(s, buildServerContext(p3ServerKey)).then(
          (x) { hsOutcome = 'success'; x.destroy(); },
          onError: (Object e) { hsOutcome = 'error ${e.runtimeType}'; },
        );
        Timer(const Duration(seconds: 1), () {
          if (how == 'destroy') { s.destroy(); } else { unawaited(s.close()); }
        });
      });
      final c = await Socket.connect('127.0.0.1', raw.port);
      c.add([0x16, 0x03, 0x01, 0x02, 0x00]);
      await c.flush();
      final closed = await clientSeesClose(c, const Duration(seconds: 4));
      // ignore: avoid_print
      print('E1[$how] client saw the server close within 4 s: $closed; handshake future: $hsOutcome');
      c.destroy();
    });
  }

  test('E5 RawServerSocket + RawSecureSocket.secureServer, raw.close() after 1 s', () async {
    final rs = await RawServerSocket.bind(InternetAddress.loopbackIPv4, 0);
    addTearDown(rs.close);
    var hsOutcome = 'pending';
    rs.listen((raw) {
      RawSecureSocket.secureServer(raw, buildServerContext(p3ServerKey)).then(
        (x) { hsOutcome = 'success'; x.close(); },
        onError: (Object e) { hsOutcome = 'error ${e.runtimeType}'; },
      );
      Timer(const Duration(seconds: 1), () => raw.close());
    });
    final c = await Socket.connect('127.0.0.1', rs.port);
    c.add([0x16, 0x03, 0x01, 0x02, 0x00]);
    await c.flush();
    final closed = await clientSeesClose(c, const Duration(seconds: 4));
    // ignore: avoid_print
    print('E5 client saw the server close within 4 s: $closed; handshake future: $hsOutcome');
    c.destroy();
  });

  test('E6 stalled handshake that is NOT touched: does dart close it by itself within 12 s?', () async {
    final raw = await ServerSocket.bind(InternetAddress.loopbackIPv4, 0);
    addTearDown(raw.close);
    var hsOutcome = 'pending';
    raw.listen((s) {
      SecureSocket.secureServer(s, buildServerContext(p3ServerKey)).then(
        (x) { hsOutcome = 'success'; x.destroy(); },
        onError: (Object e) { hsOutcome = 'error ${e.runtimeType}'; },
      );
    });
    final c = await Socket.connect('127.0.0.1', raw.port);
    c.add([0x16, 0x03, 0x01, 0x02, 0x00]);
    await c.flush();
    final closed = await clientSeesClose(c, const Duration(seconds: 12));
    // ignore: avoid_print
    print('E6 untouched: client saw close within 12 s: $closed; handshake future: $hsOutcome');
    c.destroy();
  });
}
